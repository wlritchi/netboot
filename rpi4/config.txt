# Raspberry Pi 4 boot configuration for UEFI netboot
# This config tells the Pi to boot in 64-bit mode and use RPI_EFI.fd as the firmware

# Enable 64-bit mode
arm_64bit=1

# Use RPI_EFI.fd as the UEFI firmware stub
armstub=RPI_EFI.fd

# Enable UART for debugging
enable_uart=1

# Enable GIC (Generic Interrupt Controller)
enable_gic=1

# Disable GPU kernel command line tags (required for UEFI)
disable_commandline_tags=2

# No boot delay
boot_delay=0

# Tell GPU firmware where to place the device tree blob
# Must match PcdFdtBaseAddress in EDK2 (0x003e0000)
device_tree_address=0x3e0000
device_tree_end=0x400000
